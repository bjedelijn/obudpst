cmake_minimum_required(VERSION 3.15)
project(udpst_win32_platform_smoke C)
if(NOT WIN32)
    message(FATAL_ERROR "Windows platform smoke test requires WIN32")
endif()
add_executable(udpst_win32_platform_smoke platform_smoke.c)
target_compile_definitions(udpst_win32_platform_smoke PRIVATE _WIN32_WINNT=0x0602 WIN32_LEAN_AND_MEAN)
target_link_libraries(udpst_win32_platform_smoke PRIVATE ws2_32)
enable_testing()
add_test(NAME udpst_win32_platform_smoke COMMAND udpst_win32_platform_smoke)
